```java
import com.acme.common.PageMeta;
import org.springframework.web.bind.annotation.GetMapping;

    private final InvoiceService invoiceService;

    }

    @GetMapping("/invoices")
    public ApiResponse<?> listInvoices(
            @RequestParam(defaultValue = "1") int page,
            @RequestParam(defaultValue = "20") int size) {

        int pageSize = Math.clamp(size, 1, MAX_PAGE_SIZE);

        InvoiceQueryResult result = invoiceService.listInvoices(pageNo, pageSize);

        return ApiResponse.ok(result.items(), PageMeta.of(pageNo, pageSize, result.total()));
    }
}
```

In a layered Spring Boot architecture, **the Service layer owns business rules, invariants, and error boundaries**, rather than delegating rule enforcement to repository query filters (which would conflate permission violations with non-existent entities).

### Architectural Overview

1. **Existence Verification**: Queries `InvoiceRepository.findById(invoiceId)`. If absent, raises `InvoiceNotFoundException` (mapped to **404 Not Found**).
2. **Ownership Enforcement**: Validates that `invoice.customerId() == customerId`. If mismatched, raises `InvoiceAccessDeniedException` (mapped to **403 Forbidden**).
3. **Lifecycle / State Enforcement**: Validates that the invoice is still `UNPAID` (`invoice.status() == InvoiceStatus.UNPAID`). If not, raises `InvoiceNotUnpaidException` (mapped to **409 Conflict**).
4. **State Transition**: Invokes `invoice.markCancelled()` within transaction boundaries (`@Transactional`).
5. **API Error Response Translation**: An `@RestControllerAdvice` translates domain exceptions into standard RFC 7807 `ProblemDetail` responses, while `@ResponseStatus` annotations provide direct fallback status mapping.

### 1. `InvoiceService.java`

```java
package com.acme.billing;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class InvoiceService {

    private final InvoiceRepository invoiceRepository;

    public InvoiceService(InvoiceRepository invoiceRepository) {
        this.invoiceRepository = invoiceRepository;
    }

    /**
     * Cancels an invoice if the requesting customer is the owner and the invoice is still unpaid.
     *
     * @param invoiceId  the ID of the invoice to cancel
     * @param customerId the ID of the customer requesting cancellation
     * @throws InvoiceNotFoundException     if the invoice does not exist
     * @throws InvoiceAccessDeniedException if the invoice does not belong to the requesting customer
     * @throws InvoiceNotUnpaidException    if the invoice is not in UNPAID status
     */
    @Transactional
    public void cancelInvoice(long invoiceId, long customerId) {
        Invoice invoice = invoiceRepository.findById(invoiceId)
                .orElseThrow(() -> new InvoiceNotFoundException(invoiceId));

        if (invoice.customerId() != customerId) {
            throw new InvoiceAccessDeniedException(invoiceId, customerId);
        }

        if (invoice.status() != InvoiceStatus.UNPAID) {
            throw new InvoiceNotUnpaidException(invoiceId, invoice.status());
        }

        invoice.markCancelled();
    }

    /**
     * Convenience alias for cancelInvoice.
     */
    @Transactional
    public void cancel(long invoiceId, long customerId) {
        cancelInvoice(invoiceId, customerId);
    }
}
```

```java
package com.acme.billing;

import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.net.URI;
import java.time.Instant;

@RestControllerAdvice
public class InvoiceExceptionHandler {

    @ExceptionHandler(InvoiceNotFoundException.class)
    public ProblemDetail handleInvoiceNotFound(InvoiceNotFoundException ex) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.NOT_FOUND, ex.getMessage());
        problem.setTitle("Invoice Not Found");
        problem.setType(URI.create("urn:problem-type:invoice-not-found"));
        problem.setProperty("timestamp", Instant.now());
        problem.setProperty("invoiceId", ex.getInvoiceId());
        return problem;
    }

    @ExceptionHandler(InvoiceAccessDeniedException.class)
    public ProblemDetail handleInvoiceAccessDenied(InvoiceAccessDeniedException ex) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.FORBIDDEN, ex.getMessage());
        problem.setTitle("Invoice Access Denied");
        problem.setType(URI.create("urn:problem-type:invoice-access-denied"));
        problem.setProperty("timestamp", Instant.now());
        problem.setProperty("invoiceId", ex.getInvoiceId());
        problem.setProperty("customerId", ex.getCustomerId());
        return problem;
    }

    @ExceptionHandler(InvoiceNotUnpaidException.class)
    public ProblemDetail handleInvoiceNotUnpaid(InvoiceNotUnpaidException ex) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.CONFLICT, ex.getMessage());
        problem.setTitle("Invoice Not Unpaid");
        problem.setType(URI.create("urn:problem-type:invoice-not-unpaid"));
        problem.setProperty("timestamp", Instant.now());
        problem.setProperty("invoiceId", ex.getInvoiceId());
        problem.setProperty("currentStatus", ex.getCurrentStatus());
        return problem;
    }

    @ExceptionHandler(IllegalStateException.class)
    public ProblemDetail handleIllegalState(IllegalStateException ex) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.CONFLICT, ex.getMessage());
        problem.setTitle("Conflict");
        problem.setType(URI.create("urn:problem-type:conflict"));
        problem.setProperty("timestamp", Instant.now());
        return problem;
    }
}
```
